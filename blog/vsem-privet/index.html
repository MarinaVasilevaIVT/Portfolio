                                        import requests
                                        import PySimpleGUI as sg
                                        import html

                                        # Получение вопросов из API
                                        def fetch_questions(category_url):
                                            response = requests.get(category_url)
                                            data = response.json()
                                            questions = data['results']
                                            return questions

                                        # Создание интерфейса с PySimpleGUI для вопросов
                                        def create_quiz_window(questions):
                                            layout = []
                                            for q in questions:
                                                # Декодируем HTML сущности
                                                question_text = html.unescape(q['question'])
                                                correct_answer = html.unescape(q['correct_answer'])
                                                all_answers = q['incorrect_answers'] + [correct_answer]
                                                layout.append([sg.Text(question_text, size=(60, 2))])
                                                for ans in sorted(all_answers):
                                                    layout.append([sg.Radio(html.unescape(ans), group_id=question_text, key=f'{question_text}-{ans}')])
                                                layout.append([sg.Text("", size=(60, 1), key=f'result_{question_text}')])
                                                layout.append([sg.Button('Submit', key=f'submit_{question_text}')])
                                                layout.append([sg.Text("_" * 60)])

                                            layout.append([sg.Button('Finish')])
                                            
                                            quiz_column = [[sg.Column(layout, scrollable=True, vertical_scroll_only=True, size=(600, 400))]]
                                            
                                            return sg.Window('Trivia Quiz', quiz_column, resizable=True)

                                        # Создание приветственного окна
                                        def create_welcome_window(theme = "BrightColors"):
                                            sg.theme(theme)
                                            layout = [
                                                [sg.Text("Добро пожаловать на квиз! Представься:")],
                                                [sg.InputText(key='-NAME-')],
                                                [sg.Text("Ты хочешь пройти квест?")],
                                                [sg.Button("Да :)"), sg.Button("Нет :(")]
                                            ]
                                            return sg.Window('Welcome', layout)

                                        # Создание окна выбора категории теста
                                        def create_category_window(name):
                                            layout = [
                                                [sg.Text(f"{name}, какой тест ты хочешь пройти?")],
                                                [sg.Button("Аниме"), sg.Button("Фильмы"), sg.Button("Книги")]
                                            ]
                                            return sg.Window('Select Category', layout)

                                        # Создание окна с результатами
                                        def create_results_window(correct, total):
                                            layout = [
                                                [sg.Text(f"Вы завершили тест!")],
                                                [sg.Text(f"Правильных ответов: {correct} из {total}")],
                                                [sg.Button("OK")]
                                            ]
                                            return sg.Window('Results', layout)

                                        def main():
                                            welcome_window = create_welcome_window()
                                            event, values = welcome_window.read()
                                            name = values['-NAME-'] if event == "Да :)" else None
                                            welcome_window.close()

                                            if event == "Да :)":
                                                category_window = create_category_window(name)
                                                event, values = category_window.read()
                                                category_window.close()

                                                if event == "Аниме":
                                                    category_url = "https://opentdb.com/api.php?amount=10&category=31&difficulty=easy"
                                                elif event == "Фильмы":
                                                    category_url = "https://opentdb.com/api.php?amount=10&category=11&difficulty=easy"
                                                elif event == "Книги":
                                                    category_url = "https://opentdb.com/api.php?amount=10&category=10&difficulty=easy"
                                                else:
                                                    sg.popup("Категория не выбрана. Программа завершена.")
                                                    return

                                                questions = fetch_questions(category_url)
                                                quiz_window = create_quiz_window(questions)
                                                
                                                correct_answers = 0
                                                total_questions = len(questions)

                                                while True:
                                                    event, values = quiz_window.read()
                                                    if event == sg.WIN_CLOSED or event == 'Finish':
                                                        break

                                                    for q in questions:
                                                        question_text = html.unescape(q['question'])
                                                        correct_answer = html.unescape(q['correct_answer'])
                                                        submit_event = f'submit_{question_text}'

                                                        if event == submit_event:
                                                            selected_answer = [key.split('-')[1] for key in values if values[key] and key.startswith(question_text)]
                                                            if selected_answer:
                                                                if selected_answer[0] == correct_answer:
                                                                    quiz_window[f'result_{question_text}'].update('Correct!', text_color='green')
                                                                    correct_answers += 1
                                                                else:
                                                                    quiz_window[f'result_{question_text}'].update(f'Wrong! Correct answer: {correct_answer}', text_color='red')

                                                quiz_window.close()

                                                # Показ результатов
                                                results_window = create_results_window(correct_answers, total_questions)
                                                event, values = results_window.read()
                                                if event == "OK" or event == sg.WIN_CLOSED:
                                                    results_window.close()
                                            else:
                                                sg.popup("Спасибо, до встречи!")

                                        if __name__ == "__main__":
                                            main()
